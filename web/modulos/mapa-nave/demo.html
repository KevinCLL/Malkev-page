    Cómo se conecta a la nave 3D

    // mapa-nave.js · módulo ES sin dependencias
import { init } from './mapa-nave.js';
const layout = await (await fetch('layout.json')).json();
const mapa = init(document.getElementById('mini'), layout);

// en el bucle de render
mapa.setPlayer(player.x, player.z, player.yaw);

// al tocar una estancia en el plano
mapa.onTeleport(({ x, z, yaw }) => {
  player.x = x; player.z = z; player.yaw = yaw;
});

// visitantes del multijugador (yaw de la nave)
mapa.setOthers(mj.sample().map(p => ({ ...p, yaw: p.yaw + Math.PI })));

// al salir
mapa.dispose();

    Usa el mismo layout.json que la nave, ventanales incluidos: metros, yaw 0 mirando a −Z y la proa hacia +X.
